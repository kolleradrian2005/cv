\newcommand{\tlentry}[4]{%
  \par\vspace{4pt}%

  {\bfseries\color{ink}#2}\ifx&#3&\else\hspace{4pt}{\footnotesize\color{mute}#3}\fi\par\vspace{2pt}
  #4
  \end{minipage}\par\vspace{2pt}}

% Timeline environment: resets counter, sets section id, draws rail at end
\newenvironment{timeline}[1]
  {\setcounter{tlmark}{0}%
   \renewcommand{\tlsection}{#1}%
   \hspace{\rrail}\begin{minipage}[t]{\dimexpr\linewidth-\rrail\relax}}
  {\begin{tikzpicture}[remember picture, overlay]
     \ifnum\value{tlmark}>1
       \draw[ink, line width=0.9pt]
         ($(pic cs:tl-\tlsection-1)+(-\rrail,\rdoty)$) --
         ($(pic cs:tl-\tlsection-\arabic{tlmark})+(-\rrail,\rdoty)$);
     \fi
     \foreach \i in {1,...,\arabic{tlmark}}{%
       \fill[ink]   ($(pic cs:tl-\tlsection-\i)+(-\rrail,\rdoty)$) circle (3.6pt);
       \fill[white] ($(pic cs:tl-\tlsection-\i)+(-\rrail,\rdoty)$) circle (1.7pt);}
   \end{tikzpicture}%
   \end{minipage}}
